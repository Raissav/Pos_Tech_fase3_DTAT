import sys

from awsglue.utils import getResolvedOptions
from pyspark.context import SparkContext
from awsglue.context import GlueContext
from awsglue.job import Job
from awsgluedq.transforms import EvaluateDataQuality


# ============================================================
# 1. INICIALIZAÇÃO DO JOB
# ============================================================

args = getResolvedOptions(sys.argv, ["JOB_NAME"])

sc = SparkContext()
glueContext = GlueContext(sc)
spark = glueContext.spark_session

job = Job(glueContext)
job.init(args["JOB_NAME"], args)


# ============================================================
# 2. REGRAS DE QUALIDADE DOS DADOS
# ============================================================

DEFAULT_DATA_QUALITY_RULESET = """
    Rules = [
        ColumnCount > 0
    ]
"""


# ============================================================
# 3. CONFIGURAÇÕES DE LEITURA DOS CSVs
# ============================================================

CSV_FORMAT_OPTIONS = {
    "quoteChar": "\"",
    "withHeader": True,
    "separator": ",",
    "optimizePerformance": False
}


# ============================================================
# 4. LEITURA DOS DADOS DE 2023
# ============================================================

dados_2023 = glueContext.create_dynamic_frame.from_options(
    connection_type="s3",
    connection_options={
        "paths": [
            "s3://lab-223232422954/data-input/State_of_data_BR_2023_Kaggle - df_survey_2023.csv"
        ],
        "recurse": True
    },
    format="csv",
    format_options=CSV_FORMAT_OPTIONS,
    transformation_ctx="dados_2023"
)


# ============================================================
# 5. LEITURA DOS DADOS DE 2024
# ============================================================

dados_2024 = glueContext.create_dynamic_frame.from_options(
    connection_type="s3",
    connection_options={
        "paths": [
            "s3://lab-223232422954/data-input/Final Dataset - State of Data 2024 - Kaggle - df_survey_2024.csv"
        ],
        "recurse": True
    },
    format="csv",
    format_options=CSV_FORMAT_OPTIONS,
    transformation_ctx="dados_2024"
)


# ============================================================
# 6. LEITURA DOS DADOS DE 2025-2026
# ============================================================

dados_2025_2026 = glueContext.create_dynamic_frame.from_options(
    connection_type="s3",
    connection_options={
        "paths": [
            "s3://lab-223232422954/data-input/Final Dataset - State of Data 2025-2026 - Kaggle.csv"
        ],
        "recurse": True
    },
    format="csv",
    format_options=CSV_FORMAT_OPTIONS,
    transformation_ctx="dados_2025_2026"
)


# ============================================================
# 7. AVALIAÇÃO DA QUALIDADE DOS DADOS
# ============================================================

avaliacao_2023 = EvaluateDataQuality().process_rows(
    frame=dados_2023,
    ruleset=DEFAULT_DATA_QUALITY_RULESET,
    publishing_options={
        "dataQualityEvaluationContext": "DQ_StateOfData_2023",
        "enableDataQualityResultsPublishing": True
    },
    additional_options={
        "dataQualityResultsPublishing.strategy": "BEST_EFFORT",
        "observations.scope": "ALL"
    }
)

avaliacao_2024 = EvaluateDataQuality().process_rows(
    frame=dados_2024,
    ruleset=DEFAULT_DATA_QUALITY_RULESET,
    publishing_options={
        "dataQualityEvaluationContext": "DQ_StateOfData_2024",
        "enableDataQualityResultsPublishing": True
    },
    additional_options={
        "dataQualityResultsPublishing.strategy": "BEST_EFFORT",
        "observations.scope": "ALL"
    }
)

avaliacao_2025_2026 = EvaluateDataQuality().process_rows(
    frame=dados_2025_2026,
    ruleset=DEFAULT_DATA_QUALITY_RULESET,
    publishing_options={
        "dataQualityEvaluationContext": "DQ_StateOfData_2025_2026",
        "enableDataQualityResultsPublishing": True
    },
    additional_options={
        "dataQualityResultsPublishing.strategy": "BEST_EFFORT",
        "observations.scope": "ALL"
    }
)


# ============================================================
# 8. GRAVAÇÃO DOS DADOS EM PARQUET NO S3
# ============================================================

# Dados de 2023
glueContext.write_dynamic_frame.from_options(
    frame=dados_2023,
    connection_type="s3",
    format="glueparquet",
    connection_options={
        "path": "s3://lab-223232422954/data-output/ano=2023/",
        "partitionKeys": []
    },
    format_options={
        "compression": "snappy"
    },
    transformation_ctx="saida_parquet_2023"
)


# Dados de 2024
glueContext.write_dynamic_frame.from_options(
    frame=dados_2024,
    connection_type="s3",
    format="glueparquet",
    connection_options={
        "path": "s3://lab-223232422954/data-output/ano=2024/",
        "partitionKeys": []
    },
    format_options={
        "compression": "snappy"
    },
    transformation_ctx="saida_parquet_2024"
)


# Dados de 2025-2026
glueContext.write_dynamic_frame.from_options(
    frame=dados_2025_2026,
    connection_type="s3",
    format="glueparquet",
    connection_options={
        "path": "s3://lab-223232422954/data-output/ano=2025-2026/",
        "partitionKeys": []
    },
    format_options={
        "compression": "snappy"
    },
    transformation_ctx="saida_parquet_2025_2026"
)


# ============================================================
# 9. FINALIZAÇÃO DO JOB
# ============================================================

job.commit()
